One call to deposit, one to check the live balance. The SDK handles chains, gas and routing for you.

// npm i @bloomfi/sdk import { Bloom } from "@bloomfi/sdk"; const bloom = new Bloom(process.env.BLOOM_KEY); // deposit 500 USDC into USD bloom await bloom.deposit({ amount: "500", asset: "USDC", chain: "base", }); const b = await bloom.balance(); console.log(b.apy); // 8.4
